"""probe_iff_67d — task 67-d property probe, written FIRST (before any fix on this lane).

Property under test (mission item 1):
    For every cell c within radius r of the origin,
        hex_distance(origin, c) == 1  <=>  c in origin.neighbors()

Run against any checkout:
    PYTHONPATH=<repo>/ python3 probe_iff_67d.py [radius]
or as pytest (copy into the checkout's tests/ dir):
    python3 -m pytest tests/probe_iff_67d.py -v
"""

import os
import sys

from slackwater_lattice.eisenstein import EisensteinInteger, hex_distance

ORIGIN = EisensteinInteger(0, 0)
RADIUS = int(os.environ.get("PROBE_RADIUS", "4"))


def ball(radius):
    out = []
    for da in range(-radius, radius + 1):
        for db in range(-radius, radius + 1):
            p = EisensteinInteger(da, db)
            if hex_distance(ORIGIN, p) <= radius:
                out.append(p)
    return out


def iff_violations(radius):
    cells = ball(radius)
    violations = []
    for c in cells:
        is_neighbor = c in set(ORIGIN.neighbors())
        scored_one = hex_distance(ORIGIN, c) == 1
        if is_neighbor != scored_one:
            violations.append(
                (c, "neighbor scored != 1" if is_neighbor else "non-neighbor scored 1", hex_distance(ORIGIN, c))
            )
    return cells, violations


def test_iff_holds_within_radius_of_origin():
    cells, violations = iff_violations(RADIUS)
    assert not violations, (
        f"{len(violations)} of {len(cells)} cells violate "
        f"hex_distance(origin,c)==1 <=> c in neighbors(origin); "
        f"first witnesses: {violations[:6]}"
    )


if __name__ == "__main__":
    cells, violations = iff_violations(RADIUS)
    print(f"probe_iff_67d: radius={RADIUS}, cells={len(cells)}, iff-violations={len(violations)}")
    for v in violations[:10]:
        print(f"  VIOLATION: cell={v[0]} class={v[1]} hex_distance={v[2]}")
    print("RESULT:", "FAIL" if violations else "PASS")
    sys.exit(1 if violations else 0)
